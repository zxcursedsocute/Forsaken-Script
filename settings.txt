local a,b,c=...
local d=game:GetService"HttpService"
local e=false

local f="http://zxccursed.temp.swtest.ru/Scripts/Save_config.php"
local g=(syn and syn.request)or(http and http.request)or request

local function notify(h,i,j,k)
b:Notification{Name=h,Description=i,Type=j,Duration=k}
end

local function getCfgPath()
return b.saveFolder.."/"..b.saveFileName
end

local function saveCfg()
b:SaveConfig(b.saveFolder,b.saveFileName)
end

a:AddSection"Interface"
a:AddDropdown{Name="Theme",Description="Change the interface theme",Options={'Two Time Femboy (by Mary)','Skitty (by Cozy)','Jane Doe (by Entity)',"Forest 2021","Cursed",'Darkness','Dark','White','Black'},Callback=function(h)b:SetTheme(h)end}
a:AddToggle{Name="Transparency",Description="Change The Background Transparency",Callback=function()b:ChangeBackgroundTransparance()end}
a:AddToggle{Name="User info",Description="Show info about your account",Callback=function()b:UserInfo()end}


if c.KeyboardEnabled then
a:AddSection"Window"
a:AddKeybind{Name="Minimize Window",Description="Change the window to minimize",Callback=function()b:Minimaze()end}
a:AddKeybind{Name="Column Window",Description="Change the window to column",Callback=function()b:ColumnWindow()end}
a:AddKeybind{Name="Close Window",Description="Close the Window",Default=Enum.KeyCode.LeftAlt,Callback=function()b:Close()end}
end


a:AddSection"Cloud Configs"

local c=""

a:AddInput{
Name="Load Cloud Config",
Description="Cloud ID (Load) / Name (Upload)",
Callback=function(d)
c=d or""
local e=tostring(d or""):gsub("%s+","")

if e~=""and tonumber(e)then
if not http_request then return notify("Error","Your executor does not support HTTP requests!")end

local f=(CLOUD_URL or"http://zxccursed.temp.swtest.ru/Scripts/Save_config.php").."?action=download&id="..e
local g=http_request{
Url=f,
Method="GET"
}

if g.StatusCode==200 and g.Body~=""then
local h="cloud_"..e..".json"
local i=tostring(b.saveFolder).."/"..h
writefile(i,g.Body)

b:UpdateConfigSelection(h)
if cfgDrop then cfgDrop:UpdateOptions(b:GetConfigsList())end

if b:LoadConfig(i)then
notify("Cloud Success","Config ID "..e.." successfully loaded!",nil,4)
else
notify("Error","Error reading downloaded config.")
end
elseif g.StatusCode==404 then
notify("Error","Config with ID "..e.." not found!")
else
notify("Error","Download error: HTTP "..tostring(g.StatusCode))
end
else
if d~=""then
notify("Cloud Info","Name set to: '"..tostring(d).."'\nNow click Upload!",nil,3)
end
end
end
}

a:AddButton{
Name="Upload Config to Cloud",
Description="Saves current config to server",
Callback=function()
local d=getCfgPath()
if not isfile(d)then return notify("Error","No saved config to upload!")end
if not http_request then return notify("Error","Your executor does not support HTTP requests!")end

local e=readfile(d)
local f=(c~=""and c)or tostring(b.saveFileName or"config")


local g=(f:gsub("([^%w %-%_%.%~])",function(g)
return string.format("%%%02X",string.byte(g))
end):gsub(" ","+"))

local h=(CLOUD_URL or"http://zxccursed.temp.swtest.ru/Scripts/Save_config.php").."?action=upload&name="..g
local i=http_request{
Url=h,
Method="POST",
Headers={["Content-Type"]="application/json"},
Body=e
}

if i.StatusCode==200 then
local j=hs or game:GetService"HttpService"
local k,l=pcall(function()return j:JSONDecode(i.Body)end)
if k and type(l)=="table"and l.id then
setclipboard(tostring(l.id))
notify("Cloud Success","Uploaded as '"..tostring(l.name or f).."'\nID copied to clipboard!",nil,5)
else
notify("Error","Invalid response from server")
end
elseif i.StatusCode==429 then
notify("Cooldown","You can only upload a config once per 1 min!",nil,5)
else
notify("Error","Upload error: HTTP "..tostring(i.StatusCode))
end
end
}

end

}

a:AddSection"Advanced Config"

a:AddToggle{
Name="Auto Load Config",
Callback=function(i)
local j=getCfgPath()
local k={Toggles={}}

if isfile(j)then
local l,m=pcall(function()return d:JSONDecode(readfile(j))end)
if l and type(m)=="table"then
k=m
k.Toggles=k.Toggles or{}
end
end

k.Toggles["Auto Load Config"]=i
if not isfolder(b.saveFolder)then makefolder(b.saveFolder)end

local l,m=pcall(function()return d:JSONEncode(k)end)
if l then writefile(j,m)end

if i and isfile(j)then
b:LoadConfig(j)
notify("Success","Config "..b.saveFileName.." applied!",nil,3)
end
end,
}

a:AddToggle{
Name="Auto Save Config",
Callback=function(i)
e=i
if not i then return notify("Info","Auto Save disabled.",nil,3)end

saveCfg()
notify("Success","Auto Save enabled. Saving every 10s.",nil,3)

task.spawn(function()
while e do
task.wait(10)
if e then saveCfg()end
end
end)
end
